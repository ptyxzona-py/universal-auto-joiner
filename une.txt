-- Fetch available servers from the game
function ServerJoiner:fetchServers()
    local servers = {}
    
    pcall(function()
        local response = HttpService:GetAsync("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public? sortOrder=Asc&limit=100")
        local data = HttpService:JSONDecode(response)
        
        if data. data then
            for _, server in ipairs(data.data) do
                table.insert(servers, {
                    id = server.id,
                    playerCount = server.playerCount,
                    maxPlayers = server.maxPlayers
                })
            end
        end
    end)
    
    return servers
end

-- Display servers in GUI
function ServerJoiner:displayServers(serverList, frame)
    frame.CanvasSize = UDim2.new(0, 0, 0, #serverList * 30)
    
    for _, child in ipairs(frame: GetChildren()) do
        child: Destroy()
    end
    
    for i, server in ipairs(serverList) do
        local serverButton = Instance.new("TextButton")
        serverButton.Name = "Server_" .. i
        serverButton. Size = UDim2.new(1, -10, 0, 25)
        serverButton.Position = UDim2.new(0, 5, 0, (i-1) * 30)
        serverButton.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
        serverButton.TextColor3 = Color3.fromRGB(200, 255, 200)
        serverButton.TextSize = 11
        serverButton.Text = "Server ID: " .. server.id ..  " | Players: " .. server.playerCount ..  "/" .. server.maxPlayers
        serverButton.Font = Enum. Font.Gotham
        serverButton.Parent = frame
        
        serverButton.MouseButton1Click:Connect(function()
            ServerJoiner:joinServer(server.id)
        end)
    end
end

-- Join a specific server
function ServerJoiner:joinServer(serverId)
    pcall(function()
        TeleportService:TeleportToPlaceInstance(game.PlaceId, serverId, Players.LocalPlayer)
    end)
end

-- Send message in chat
function ServerJoiner:sendMessage(message)
    pcall(function()
        local textChatService = game:GetService("TextChatService")
        if textChatService and textChatService.ChatInputTextChannel then
            textChatService. ChatInputTextChannel:SendAsync(message)
        end
    end)
end

-- Start the loop
function ServerJoiner:start(statusLabel, serverListFrame)
    self.isRunning = true
    
    while self.isRunning do
        local servers = self:fetchServers()
        
        if #servers > 0 then
            self: displayServers(servers, serverListFrame)
            
            -- Join a random server
            local randomServer = servers[math.random(1, #servers)]
            statusLabel.Text = "Status: 🔄 Joining...\nServer: " .. randomServer.id
            
            wait(1)
            self:joinServer(randomServer.id)
            
            wait(3) -- Wait for game to load
            self:sendMessage(self.customMessage)
            
            statusLabel.Text = "Status: ✅ Message Sent!\nWaiting " .. self.repeatDelay .. "s..."
            wait(self. repeatDelay)
        else
            statusLabel.Text = "Status: ❌ No servers found"
            wait(5)
        end
    end
end

-- Stop the loop
function ServerJoiner:stop(statusLabel)
    self.isRunning = false
    statusLabel.Text = "Status: ⏹️ Stopped"
end
